## 3. 源码还原与代码骨架

```swift
// 3.1 SettingsScene（SwiftUI，页签数组数据驱动——原实现即此模式）
//     实证：页签=静态数组(6→5 补丁 I) + ForEach + 标题选择器(0x1005fd7e8)
enum SettingsTab: String, CaseIterable, Identifiable {
    case general, search, tools, shortcuts, about   // 原 enum 含 .pro（已移除）
}
struct SettingsScene: View {
    @State private var selectedTab: SettingsTab = .general
    var body: some View {
        VStack(spacing: 0) {
            SettingsNavBar(selection: $selectedTab)      // 页签栏（对齐截图样式）
            contentFrame(for: selectedTab)               // 各页内容
        }
    }
}

// 3.2 各页（重建骨架，内容按截图复刻）
struct GeneralSettingsView: View { /* 启动/外观/语言/功能/图标/监控/自动粘贴/标签/AI入口 */ }
struct SearchSettingsView: View  { /* 搜索行为配置 */ }
struct ToolsSettingsView: View   { /* 工具配置（入口→10 号） */ }
struct ShortcutsSettingsView: View { /* 快捷键表（录入控件/冲突提示） */ }
struct AboutSettingsView: View   { /* 版本/链接/致谢 + 彩蛋 */ }

// 3.3 存储（沿用原 key 命名以保持兼容——开发时用 defaults dump 比对）
enum SettingsKeys {
    static let launchAtLogin   = "launch_at_login"      // （推断名，需实机校准）
    static let appearance      = "appearance_mode"      // Midnight 外观
    static let language        = "app_language"
    // …按通用页截图逐项补齐
}
```
