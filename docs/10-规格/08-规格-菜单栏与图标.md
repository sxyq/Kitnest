## 3. 源码还原与代码骨架

```swift
// 3.1 主状态项（AppKit，原行为复刻）
final class StatusBarController {
    private var statusItem: NSStatusItem!
    func install() {
        statusItem = NSStatusBar.system.statusItem(withLength: NSStatusItem.variableLength)
        statusItem.button?.image = AppIconProvider.menuBarIcon()   // 模板图像（随主题变色）
        statusItem.button?.action = #selector(togglePanel)
        statusItem.button?.target = self
    }
    @objc private func togglePanel() { FloatingPanel.shared.toggle() }
}

// 3.2 通知条（Toast）
final class MenuBarToastManager {
    func show(_ text: String, near item: NSStatusItem) {
        // 弹出短文本气泡（图标旁），1.5s 后淡出（动画状态用 ToastAnimationState 驱动）
    }
}

// 3.3 图标提供（带缓存——与 file_type_icons 表对应）
final class FileTypeIconProvider {
    func icon(forExtension ext: String) -> NSImage {
        if let cached = cacheDB.icon(for: ext) { return cached }   // file_type_icons 表
        let handler = NSWorkspace.shared.preferredApplicationBundleID(forExtension: ext)
        let icon = NSWorkspace.shared.icon(forFileType: ext)
        cacheDB.store(ext: ext, handler: handler, icon: icon)      // 落盘缓存
        return icon
    }
}
```
